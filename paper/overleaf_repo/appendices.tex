\begin{definition}
Let $S_n$ be the set of all permutations on $n$ elements.
\end{definition}

\begin{definition}[Permutation invariance]
\label{def:permutation invariance}
A function $f:\R^n \to \R^m$ is permutation invariant iff, for all $P \in S_n$ and $\x \in \R^n$, $f(P(\x)) = f(\x)$.
% TODO: this is not how permutation invariance is defined in the zaheer paper though, why is it equivalent? this is important to be able to use theorem 10 in the current wording
\end{definition}

\begin{definition}
%Let $f:\R^n \to \R^m$.
Let $\sim_P$ be the equivalence relation
\begin{equation}
    \x \sim_P \x' \iff \exists P \in S_n : P(\x) = \x'
\end{equation}
\end{definition}

\begin{definition}
Let $[\x]_P$ be the equivalence class of $\x\in\R^n$ induced by $\sim_P$:
\begin{equation}
    \label{eq:equivalence class}
    [\x]_P=\{\x' \in \R^n: \x' \sim_P \x\}
\end{equation}
\end{definition}

\begin{definition}
Let $\R^n/\sim_P:=\{[\x]_P\}$ denote the set of all equivalence classes $[\x]_P$, i.e. the quotient space over $\R^n$ induced by $\sim_P$. 
%
\end{definition}

\begin{lemma}
\label{lemma:permutation invariance}
$f:\R^n \to \R^m$ is permutation invariant iff
%it can be factored.
there exists $b : \R^n/\sim_P \to \R^m$ such that $f = b \circ \pi$,
%where $\;\pi : \R^n \to \R^n/\sim_P$ is the canonical surjection $\x\mapsto [\x]_P$.
where $\;\pi$ is the canonical surjection $\x\mapsto [\x]_P$.
\end{lemma}

\begin{proof}
Let $P' \in S_n$ and $\x \in \R$. Now let $\x'=P'(\x)$. Since $P'$ exists, by definition $\x'\sim_P \x$ and $[\x]_P=[\x']_P$. Then 
\begin{equation}
    \label{eq:p1}
    \pi(P'(\x)) = [P'(\x)]_P
    = [\x']_P
    = [\x]_P
    = \pi(\x)
\end{equation}
To prove sufficiency, if $b$ exist and $f=b\circ \pi$, then 
\begin{align}
    f(\x) &= b(\pi(\x))\nonumber\\
    &= b([\x]_P)\nonumber\\
    &= b([P'(\x)]_P)\nonumber\\
    &= b(\pi(P'(\x)))\nonumber\\
    &= f(P'(\x)))
\end{align}
This is valid for all $\x \in \R^n$ and $P'\in S_n$, since both were taken at random. Therefore $f$ is permutation invariant.

To show necessity, let $f:\R^n \to \R^m$ be a permutation invariant function. Define $b([\x]_P)=f(\x)$. Let $\x,\x'\in \R^n$.
If $b([\x]_P) \neq b([\x']_P)$, $f(\x) \neq f(\x')$ by definition.
Since $f$ is permutation invariant, $\nexists P\in S_n : \x' = P(\x)$. Therefore, $\x'\nsim_P \x$, which implies $[\x']_P \neq [\x]_P$, showing that $b([\x]_P) \neq b([\x']_P) \implies [\x]_P \neq [\x']_P$. Since $\x$ and $\x'$ were taken at random, this shows that $b$ is a well-defined function.
\end{proof}

\begin{definition}[Surjectivity]
\[
    c\colon \mathcal{A} \to \mathcal{C} \text{ is surjective} \iff \forall C \in \mathcal{C},\exists A \in \mathcal{A} : c(A) = C
\]
\end{definition}

\begin{lemma}
\label{lemma:surj}

If the composition of two functions is surjective, the outer-most component function is also surjective:
\begin{equation}
  \begin{tikzcd}
    & \mathcal{B} \arrow{dr}{b} &
    \\
    \mathcal{A} \arrow{ur}{a} \arrow[two heads]{rr}{c} & & \mathcal{C}
  \end{tikzcd}
\end{equation}
% \[
%     \exists \text{surjective} c\colon \mathcal{A} \twoheadrightarrow \mathcal{C} \mid c = b\circ a \implies b \text{ is surjective}
% \]
\begin{multline}
    \forall a: \mathcal{A} \to \mathcal{B}, b: \mathcal{B} \to \mathcal{C}, c: \mathcal{A} \to \mathcal{C}:
    \\
    :(c = b \circ a) \land (c \text{ surjective}) \implies b \text{ surjective}
\end{multline}
\end{lemma}

\end{proof}

\begin{theorem}[\citet{zaheer_deep_2017}]
\label{the:deepsets}

% A map $\phi\colon\mathcal{A}\to \mathcal{B}$ between finite-dimension vector spaces $\mathcal{A}$ and $\mathcal{B}$ is permutation-invariant if and only if
% $\exists\; \rho,\,\mu\ \mid\forall\; A \in \mathcal{A},\;\phi(A) = \rho \left(\sum_{a \in A}\mu(a)\right)$.
Let $f\colon[0,\; 1]^M \to \mathbb{R}$ be a permutation-invariant continuous function iff it has the representation
\begin{equation}
    f\begin{pmatrix}
        x_1, & x_2, & \dots, & x_M 
    \end{pmatrix}
    = \rho\left(\sum_{m=1}^M\mu(x_m)\right)
\end{equation}
for some continuous outer and inner functions $\rho \colon \mathbb{R}^{M+1}\to \mathbb{R}$ and $\mu \colon \mathbb{R}\to \mathbb{R}^{M+1}$ respectively. The inner function $\phi$ is independent on the function $f$.

%$\exists\; \rho,\,\mu\ \mid\forall\; A \in \mathcal{A},\;\phi(A) = \rho \left(\sum_{a \in A}\mu(a)\right)$.
\end{theorem}

\begin{theorem}
\label{the:oxytrees}
There exists both a row-wise and a column-wise proxy representation for $I(Y)$ if and only if there exists $\rho$ and $\mu$ functions such that 
\begin{equation}
    I(Y)=\rho\left(\sum_{ij}\mu(Y^{ij})\right)
\end{equation}
\end{theorem}
\begin{proof}
To prove sufficiency, let 
\begin{align}
    \mu_r(z_1, \dots, z_m) &= \sum_j \mu(z_j)\\
    \rho_r(z_1, \dots, z_n) &= \rho\left(\sum_{i}z_i\right)\\
    \mu_c(z_1, \dots, z_n) &= \sum_i \mu(z_i)\\
    \rho_c(z_1, \dots, z_m) &= \rho\left(\sum_{j}z_j\right)
\end{align}
%
Given \cref{the:deepsets}, these functions follow the definition of proxies (\cref{def:proxy}).

To prove necessity, let $Y \in \Y^{n \times m}$. Let $(Y^{ij}\; Y^{kl})$ be the permutation of elements $Y^{ij}$ and $Y^{kl}$. $(Y^{ij}\; Y^{kl})$ can be written as a composition of permutations in a single column or row:
\begin{equation}
    (Y^{ij}\; Y^{kl}) = (Y^{ij}\; Y^{il}) \circ (Y^{il}\; Y^{jl}) \circ (Y^{ij}\; Y^{il})
\end{equation}
\Cref{lemma:single row permutation} implies that $I((Y^{ij}\; Y^{il})(Y)) = I(Y)$ and \cref{lemma:single col permutation} implies that $I((Y^{ij}\; Y^{kj})(Y)) = I(Y)$. Therefore
\begin{align}
I(&(Y^{ij}\; Y^{kl})(Y))=\nonumber\\
&=I((Y^{ij}\; Y^{il}) \circ (Y^{il}\; Y^{jl}) \circ (Y^{ij}\; Y^{il})(Y))\nonumber\\
&=I((Y^{ij}\; Y^{il}) \circ (Y^{il}\; Y^{jl})(Y))\nonumber\\
&=I((Y^{ij}\; Y^{il}) (Y))\nonumber\\
&=I(Y)
\end{align}
This is valid for any indices $i,j,k,l$, since they were taken at random.
Since any permutation of the elements of $Y$ can be written as a composition of permutations $(Y^{ij}\; Y^{kl})$, $I$ is invariant under any permutation of elements in $Y$. From \cref{the:deepsets}, it follows that $I$ can be expressed as
\begin{equation}
    I\begin{pmatrix}
        Y^{11}, & \dots, & Y^{nm}
    \end{pmatrix}
    = \rho\left(\sum_{j=1}^m\sum_{i=1}^n\mu(Y^{ij})\right)
\end{equation}
for some continuous $\rho \colon \mathbb{R}^{M+1}\to \mathbb{R}$ and $\mu \colon \mathbb{R}\to \mathbb{R}^{M+1}$.

\end{proof}

\begin{corollary}
The impurity followed by \citet{pliakos_global_2018} does not allow both row and proxy representations:
\begin{equation}
    \label{eq:pbct}
    I_\text{PBCT}(Y) = \frac{1}{m}\sum_j\left(\frac{1}{n}\sum_i (Y^{ij})^2 - \left(\frac{1}{n}\sum_i (Y^{ij})\right)^2\right)
\end{equation}
\end{corollary}
\begin{proof}
   From the representation in \cref{the:oxytrees}, we have that $I(Y) = I(\text{vec}(Y))$.
   Take a matrix $Y'$ for which columns are constant. From \cref{eq:pbct}, $I_\text{PBCT}(Y')=0 \neq I_\text{PBCT}(\text{vec}(Y'))$.
\end{proof}

\section{Asymptotic Computational Complexity of Biclustering Decision Trees}
\label{sec:complexity analysis}

In this appendix, we formally deduce the computational complexity expressions for the training and leaf assignment procedures of bipartite decision trees, comparing our method proposals to the previous algorithms introduced by \citet{pliakos_global_2018}.

\subsection{Training Time Complexity}
\label{sec:fit complexity analysis}

%TODO: fix n_1 or n notation
We assume that the numbers of instances in each dimension have the same order of complexity. Formally, let $n_1$ and $n_2$ be the number of rows in $X_1$ and $X_2$, respectively. We assume $n_1 \in \Theta(n_2)$.
%
From \ref{alg:find best split dt} alone, the complexity of the split search would be given by the number $|\s|$ of candidate splits times the time it takes to calculate $\Delta I$ for each of the splits: $\Theta(|\s| \;\Delta I(n_1))$. However, we must consider that it is often possible to obtain multiple values of $\Delta I$ for a single feature column $(X_a\T)^f$ much more efficiently using dynamic programming techniques. For instance, we can evaluate all split thresholds between two consecutive values of $(X_1\T)^f$ in $\Theta(\Delta I(n_1))$ time instead of $\Theta (n_1\;\Delta I (n_1))$. This is done by sorting $(X_1\T)^f$, applying the same permutation that sorted $(X_1\T)^f$ to the rows of $Y$, and iteratively obtaining $\tilde Y_1$ and $\tilde Y_2$ (\cref{eq:proxies general}) from which $\Delta I$ (\cref{eq:delta i}) can be calculated.
% (see Equations \ref{eq:i rows oxytrees} and \ref{eq:i rows pbct}).
 %
% 
Therefore, the final computational complexity depends on the number of features $m$
%of candidate features chosen to be evaluated (a hyperparameter),
and not on the total number of split candidates:
\begin{equation}
    \Theta(\text{\ref{alg:find best split dt}}) = \Theta(m \Delta I (n_1))
\end{equation}
%
In general, the complexity of $\Delta I$ is linear with respect to the number of elements of $Y$. Thus, for BICTR we have $\Delta I(n_1) \in \Theta (n_1^2)$, which results in
%
\begin{align}
    \label{eq:complexity find split gmo}
    \text{\ref{alg:find best split gmo}}
        &\in \Theta(m n_1^2)
    \text{.}
\end{align}
For Oxytrees, however, $\Delta I(n_1) \in \Theta (n_1)$, since we employ lower-dimensional proxies as described in \cref{sec:oxytrees training}. After introducing an $n_1^2$ factor to account for the time spent building the proxies, we have
\begin{align}
    \label{eq:complexity find split bgso}
    \text{\ref{alg:find best split bgso}}
        &\in \Theta(n_1^2 + m n_1)
    \text{.}
\end{align}

To estimate the complexity of the entire tree-building process, we consider the ideal scenario of a balanced decision tree.\footnote{Notice that the balanced scenario \emph{disfavors} our proposed optimization. Specifically, our method improves the complexity of the node-wise impurity calculation, which is dominated by the largest node in each level. A balanced tree results in the fastest decrease in the maximum node size per level.}
We also consider that the dimension of a split alternates at each level, so that a matrix $Y$ is separated into four equally-sized partitions after two levels. This results in the following recurrence relationship:
%
\begin{equation}
    T(n_1) = 4T\left(\frac{n_1}{2}\right) + \text{\ref{alg:find best split dt}}(n_1, m)
    \label{eq:recurrence}
\end{equation}
%
$T(n_1)$ is the time to build the tree from an $n_1$ by $n_1$ interaction matrix, and \ref{alg:find best split dt}$(n_1,\; m)$ in this case is the time taken to select a split.
The algorithm complexity of such recursive functions then follows the master theorem~\cite{cormen_introduction_2022}. The theorem gives the time complexity of a function $T$ obeying the recurrence relation $T(n) = aT(n/b) + F(n)$ and $c=\log_b a$:
%
\begin{equation}
    T(n) \in \begin{cases}
        \Theta(n^c) & \text{if } F(n) \in O(n^{c-\epsilon})\text{,} \\
        \Theta(n^c \log^{k+1} n) & \text{if } F(n) \in \Theta(n^c \log^k n)\text{,}\\
        \Theta(F(n)) & \text{if } F(n) \in \Omega(n^{c+\epsilon})
            \\&\quad\text{and }F(n)\text{ is regular.}
    \end{cases}
    \label{eq:master_theorem}
\end{equation}
%
In \cref{eq:master_theorem}, $\epsilon$ is a positive infinitesimal constant and $k$ is any non-negative integer. For the third case, we say a function $F(n)$ is regular if it satisfies the \emph{regularity condition}: $aF(n/b) \leq qF(n)$, for some constant $q < 1$ and all sufficiently large $n$~\cite{cormen_introduction_2022}.
% 
For our tree algorithms, \cref{eq:recurrence} shows that $c=2$ and $F(n)$ represents the split search function.
Applying \cref{eq:master_theorem} to \cref{eq:complexity find split bgso} and \cref{eq:complexity find split gmo} finally results in \cref{eq:complexity oxytree} and \cref{eq:complexity pbct}, respectively.


\subsection{Inference Time Complexity}
\label{sec:predict complexity analysis}
Let $|X|$ denote the number of rows in a matrix $X$.
Let $n_\text{train}$ represent the order of complexity of the number of training samples in both dimensions ($|X_\text{1, train}| \in \Theta(n_\text{train})$ and $|X_\text{2, train}| \in \Theta(n_\text{train})$). Similarly, $|X_\text{1, test}| \in \Theta(n_\text{test})$ and $|X_\text{2, test}| \in \Theta(n_\text{test})$.
For the traditional procedure of traversing the tree for each dyad individually, the complexity is given by
\begin{equation}
    \text{IndividualInference} \in \Theta(n_\text{test}^2\log\, n_\text{train})
\end{equation}
in which $n_\text{test}^2$ represents the total number of test dyads and $\log\, n_\text{train}$ represents the depth of the tree.
For the proposed batch inference procedure (\cref{sec:batch inference}), we first analyse the case in which $n_\text{test} > n_\text{train}$. The total time spent is given by
\begin{equation}
    \label{eq:time batch inference}
    \text{\ref{alg:batch assign leaves}}%(n_\text{train},\; n_\text{test})
    \in \Theta( T(n_\text{train}) + n_\text{test}^2)
    \text{,}
\end{equation}
%
in which the term $n_\text{test}^2$ represents the final leaf assignment for each test dyad (\cref{ln:final leaf assignment} of \cref{alg:batch assign leaves}) and $T$ represents the recursive procedure followed until the leaves. Similarly to the rationale behind \cref{eq:recurrence}, we consider a balanced tree in order to obtain
%
\begin{equation}
    T(n_\text{train}) = 4T\left(\frac{n_\text{train}}{2}\right) + n_\text{test}(n_\text{train})
    \text{.}
    \label{eq:recurrence predict}
\end{equation}
%
The last term corresponds to the linear process of segregating the instances in one of the dimensions (\cref{ln:x segregation} of \cref{alg:batch assign leaves}), considering that the expected number of test instances in a node will depend on the number of training instances in it.
Applying \cref{eq:master_theorem}, we see that the $n_\text{test}^2$ term of \cref{eq:time batch inference} dominates in all cases, no matter how $n_\text{test}$ asymptotically depends on $n_\text{train}$. That is,
\begin{equation}
    \label{eq:complexity batch inference n test greater}
    \text{\ref{alg:batch assign leaves}}
    %(n_\text{train},\; n_\text{test})
    \in \Omega(n_\text{test}^2)
\end{equation}
for $n_\text{test} > n_\text{train}$.
For the scenario in which $n_\text{test} < n_\text{train}$,
%
\begin{multline}
    \label{eq:time batch inference}
    \text{\ref{alg:batch assign leaves}}%(n_\text{train},\; n_\text{test}) 
    \in
        \\
        \in \Theta(
            T(n_\text{test}) 
            + 2 n_\text{test}^2 (\log\;n_\text{train} - \log\; n_\text{test}))
        )
    \text{,}
\end{multline}
%
with
%
\begin{equation}
    T(n_\text{test}) = 4T\left(\frac{n_\text{test}}{2}\right) + n_\text{test}
    \text{.}
    \label{eq:recurrence predict 2}
\end{equation}
%
$T$ corresponds again to the recursive procedure of \ref{alg:batch assign leaves}, this time considering that the number of test instances in each node approximately halves with each tree level, until there is a single test instance per node in a given level. The second term of \cref{eq:time batch inference} accounts for the remaining path that each of these instances must still traverse: $n_\text{test}^2$ is the total number of test instances and $(\log\;n^2_\text{train} - \log\; n^2_\text{test})$ is the expected length of the remaining path.
Since $n_\text{test} < n_\text{train}$, only two cases are investigated with \cref{eq:master_theorem}:
%
\begin{multline}
    \label{eq:complexity batch inforence n test lower}
    \text{\ref{alg:batch assign leaves}}%(n_\text{train},\; n_\text{test})
        \in\\
        \in \begin{cases}
            \Theta(n_\text{test}^2) & \text{if }n_\text{test} \in \Theta(n_\text{train})\text{,}\\
            \Theta(n_\text{test}^2 \log\;n_\text{train}) & \text{if }n_\text{test} \in o(n_\text{train})
            \text{.}
        \end{cases}
\end{multline}
%
Combining the conclusions of \cref{eq:complexity batch inference n test greater} and \cref{eq:complexity batch inforence n test lower}, we finally arrive at \cref{eq:complexity batch inference}.
\begin{align}
    \label{eq:complexity batch inference}
    %\text{\ref{alg:batch assign leaves}}(n_\text{train},\; n_\text{test})
    \text{\ref{alg:batch assign leaves}}
        \in \begin{cases}
            \Theta(n_\text{test}) \text{ if }n_\text{test} \in \Omega(n_\text{train})\text{,}\\
            \Theta(n_\text{test} \log(n_\text{train})) \text{ otherwise.}
        \end{cases}
\end{align}

\section{Details on Comparison Methods}
\label{sec:baselines details}

This section presents further details on the baseline methods used in this study, listed in \cref{sec:baselines}.

We included NRLMF \cite{liu_neighborhood_2016,liu_lpi-nrlmf_2017,liu_predicting_2020}, since it is, to the best of our knowledge, the most relevant matrix factorization method that can handle the fully inductive case.
It does so by regularizing the loss function to preserve neighborhoods between the original feature space and the latent space. Such neighborhoods can then be used to infer latent feature vectors (and thus predictions) for unseen instances.
NRLMF was successfully applied to several interaction prediction tasks, namely drug-target~\cite{liu_neighborhood_2016}, long non-coding RNA-protein~\cite{liu_lpi-nrlmf_2017}, and microRNA-lncRNA interactions~\cite{liu_predicting_2020}
%That is, it can predict new interactions without needing to reconstruct itself, making it applicable to all cases evaluated (Section X, EXPLANATION OF CROSS VALIDATION).

Oxytrees[Deep, YR] is included to validate our induction algorithm against BICTR~\cite{pliakos_drug-target_2020}. NRLMF is used by both methods to reconstruct the output space prior to training the forest.

Since logistic regression is a popular leaf model for model trees, we have included a baseline variant of our method using it with the GSO approach, Oxytrees[Logistic]. Furthermore, it also validates the use of RLS-Kron as the leaf model.   

Further, usual deep learning methods~\cite{huang_moltrans_2020,dehghan_tripletmultidti_2023} are not directly comparable since they are designed specifically for an application, e.g., drug-target \cite{liu2024ssldti}, RNA-disease \cite{tian2024mgcnss}. To account for neural networks, we included a multi-layer perceptron trained using the GSO approach as a baseline. 

As investigated in a previous work of ours~\cite{ilidio_fast_2024}, the semi-supervised variant of PBCT presents higher computational complexity, nonetheless it does not consistently lead to improvement in terms of performance \cite{ilidio_fast_2024}. Thus, we decide not to include it.       

The MLP architecture was selected between 10 layers of 100 neurons, 5 layers of 100 neurons, $\begin{bmatrix} 200 & 100 & 100 & 100 & 50 \end{bmatrix}$ and $\begin{bmatrix} 1024 & 512 & 256 & 128 & 64 & 32 \end{bmatrix}$, each number representing the size of each layer. The remaining hyperparameters were left as the default values of the Scikit-Learn package~\cite{pedregosa_scikit-learn_2011}, version 1.3.0.

Following the original WkNNIR study~\cite{liu_drug-target_2022}, the values evaluated for the number of neighbors used by WkNNIR were $\{1,2,3,5,7,9\}$, and for the learning rate were $\{0.1,0.2,\cdots,1\}$.

The Ridge regularization parameter of RLS-avg, RLS-Kron and BLMNII, was set to 1.
%
RLS-Kron, RLS-avg and BLMNII use a linear combination of a Gaussian network kernel (calculated from $Y$) with the original similarity kernel before training the models. We evaluated the values $\{0, 0.1, 0.25, 0.5, 0.75, 0.9, 1\}$ for the weight of this combination. The width of the Gaussian kernels was set to $1$, following the original authors.
%
The network kernel was not utilized for the RLS-Kron models in the leaves of Oxytrees.

% The Ridge regularization parameter of all linear models, including RLS-Kron, was set to 1, and no network-based kernel was used for .

NRLMF used a factor of $c=5$ to prioritize positive instances, as originally~\cite{liu_neighborhood_2016}. A maximum of 100 iterations with a minimum relative improvement of $10^{-5}$ for the loss function was performed. The number of neighbors was chosen between 3, 5 and 10, and the number of latent components was either 50 or 100. All remaining hyperparameters were randomly sampled from a log-uniform distribution between $2^{-4}$ and $2$. Due to the large hyperparameter space, 100 random combinations of hyperparameters were evaluated per outer fold for NRLMF, in contrast to the exhaustive search used with the other baselines. When used in combination with the forests, the hyperparameters were evaluated on the basis of the output of NRLMF, and not the final output of the forest. Furthermore, when combining NRLMF with forests, the original positive annotations are recovered, and only the NRLMF predictions for the negative (missing) annotations are kept~\cite{pliakos_drug-target_2020}.

All biclustering forests (BICTR and Oxytrees) utilized Extremely Randomized~\cite{geurts_extremely_2006} versions of the biclustering trees. That is, one random split candidate was selected for each candidate feature in each node. No bootstrapping of samples was performed, and all forests utilized 200 trees.
Following \cite{pliakos_network_2019,pliakos_drug-target_2020}, Oxytrees[Deep, YR] and BICTR had their trees grown until their maximum depth (reaching leaves with a single type of annotation) and evaluated $\lceil \sqrt{m}\rceil$ feature candidates in each node, from the $m$ total features. The remaining Oxytrees used minimum leaf dimensions of 5 by 5 as the only stopping criterion, and evaluated all $m$ features.

WkNNIR~\cite{liu_drug-target_2022} and BICTR~\cite{pliakos_global_2018} propose slight variations for the semi-inductive settings. We did not use them, since we primarily focus on inductive models.
